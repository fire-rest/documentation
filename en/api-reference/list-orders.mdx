---
title: "List orders"
api: "GET https://api.fire.rest/v1/orders"
description: "Returns a paginated list of orders previously injected into Fire. Filterable by store, channel, status, and date range."
authMethod: "bearer"
---

<ParamField header="x-api-key" type="string" required>
  Your Fire API key with the `orders:read` scope.
</ParamField>

<ParamField query="store_id" type="string">
  Filter by store. If omitted, returns orders across all stores the API key can see.
</ParamField>

<ParamField query="channel_id" type="string">
  Filter by sales channel.
</ParamField>

<ParamField query="status" type="string">
  Filter by order status: `OPEN`, `COMPLETED`, `CANCELLED`, `FORCE_CLOSED`.
</ParamField>

<ParamField query="payment_status" type="string">
  Filter by payment status: `PENDING`, `SUCCEEDED`, `FAILED`.
</ParamField>

<ParamField query="business_day_from" type="string">
  Lower bound (inclusive) for `businessDayDate`, format `YYYY-MM-DD`.
</ParamField>

<ParamField query="business_day_to" type="string">
  Upper bound (inclusive) for `businessDayDate`.
</ParamField>

<ParamField query="created_from" type="string">
  Lower bound (inclusive) for `createdAt`, ISO 8601 UTC.
</ParamField>

<ParamField query="created_to" type="string">
  Upper bound (inclusive) for `createdAt`.
</ParamField>

<ParamField query="limit" type="integer" default="50">
  Page size. Max `200`.
</ParamField>

<ParamField query="starting_after" type="string">
  Cursor returned in the previous response's `next_cursor`. Omit for the first page.
</ParamField>

### Response

<ResponseField name="data" type="object[]">
  Array of orders, newest first. Each item carries the full V4 Order Snapshot — same shape as the `data` block in the [`order.completed`](/en/webhook-reference/order-completed) event.
</ResponseField>

<ResponseField name="has_more" type="boolean">
  `true` if more orders match the filters beyond this page.
</ResponseField>

<ResponseField name="next_cursor" type="string | null">
  Pass this value as `starting_after` on the next request. `null` when `has_more` is `false`.
</ResponseField>
