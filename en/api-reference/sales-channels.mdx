---
title: "List sales channels"
api: "GET https://api.fire.rest/v1/sales-channels"
description: "Returns the sales channels configured in Fire. Use channel IDs when injecting orders."
authMethod: "bearer"
---

<ParamField query="store_id" type="string">
  Filter channels by store. If omitted, returns all channels across all stores.
</ParamField>

<ResponseField name="channels" type="object[]">
  List of sales channels.

  <Expandable title="channels[n]">
    <ResponseField name="id" type="string">
      Fire channel ID. Use this value as `channelId` when injecting orders.
    </ResponseField>
    <ResponseField name="name" type="string">
      Channel display name.
    </ResponseField>
    <ResponseField name="type" type="string">
      Aggregator type identifier (e.g. `uber_eats`, `rappi`, `app`, `kiosk`).
    </ResponseField>
    <ResponseField name="store_id" type="string">
      The store this channel belongs to.
    </ResponseField>
    <ResponseField name="active" type="boolean">
      Whether the channel is currently active.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseExample>
```json 200
{
  "channels": [
    {
      "id": "channel_abc123",
      "name": "Uber Eats",
      "type": "uber_eats",
      "store_id": "store_xyz",
      "active": true
    },
    {
      "id": "channel_def456",
      "name": "Rappi",
      "type": "rappi",
      "store_id": "store_xyz",
      "active": true
    }
  ]
}
```

```json 401
{
  "error": {
    "code": "unauthorized",
    "message": "Invalid or missing API key"
  }
}
```
</ResponseExample>
