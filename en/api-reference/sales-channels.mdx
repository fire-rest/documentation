---
title: "List sales channels"
api: "GET /api/v4/integrations/restaurant-os/channels"
description: "Returns the sales channels configured in Fire. Use channel IDs when injecting orders."
authMethod: "bearer"
---

<ParamField header="x-api-key" type="string" required>
  Your Fire API key.
</ParamField>

<ParamField header="x-client-channel" type="string" required>
  Must be `integration`. Identifies the request as coming from an external integration.
</ParamField>

<ParamField header="account" type="string" required>
  Account identifier the request belongs to.
</ParamField>

<ParamField query="store_id" type="string">
  Filter channels by store. If omitted, returns all channels across all stores.
</ParamField>

<ResponseField name="channels" type="object[]">
  List of sales channels.

  <Expandable title="channels[n]">
    <ResponseField name="channelId" type="string">
      Unique channel identifier. Use this value as `channelId` when injecting orders.
    </ResponseField>
    <ResponseField name="code" type="string">
      Short code for the channel (e.g. `IFOOD`, `RAPPI`, `KIOSK`). Stable identifier for programmatic mapping.
    </ResponseField>
    <ResponseField name="name" type="string">
      Channel display name.
    </ResponseField>
    <ResponseField name="description" type="string">
      Optional description of the channel.
    </ResponseField>
    <ResponseField name="active" type="string">
      Channel status: `ACTIVE` or `INACTIVE`.
    </ResponseField>
    <ResponseField name="authType" type="string">
      Authentication mechanism used by this channel's integration: `API_KEY`, `OAUTH2`, `BASIC_AUTH`, `XMART_LOGIN`, or `NONE`.
    </ResponseField>
    <ResponseField name="baseUrl" type="string">
      Base URL for the channel's integration endpoint. `null` if not applicable.
    </ResponseField>
    <ResponseField name="services" type="object[]">
      Services available through this channel.

      <Expandable title="services[n]">
        <ResponseField name="code" type="string">
          Service type code (e.g. `DELIVERY`, `TAKEAWAY`).
        </ResponseField>
        <ResponseField name="serviceId" type="string">
          Unique service identifier.
        </ResponseField>
        <ResponseField name="referenceName" type="string">
          Human-readable service name for this channel.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="vendorId" type="integer">
      Brand identifier this channel belongs to.
    </ResponseField>
    <ResponseField name="store_id" type="string">
      The store this channel is associated with.
    </ResponseField>
    <ResponseField name="createdAt" type="string">
      ISO 8601 timestamp when the channel was created.
    </ResponseField>
    <ResponseField name="updatedAt" type="string">
      ISO 8601 timestamp of the last update.
    </ResponseField>
  </Expandable>
</ResponseField>
