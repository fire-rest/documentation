---
title: "List transactions"
api: "GET https://boh.api.fire.rest/api/v1/public/operations/transactions"
description: "Lists recent inventory transactions for the account."
authMethod: "none"
---

<ParamField header="x-api-key" type="string" required>
  API key with `inventory:read` scope.
</ParamField>

<ParamField query="limit" type="number">
  Results per page (1–200, default 50).
</ParamField>

<ParamField query="offset" type="number">
  Pagination offset (default 0).
</ParamField>

<ParamField query="status" type="string">
  Filter by status: `PENDING`, `PROCESSING`, `COMPLETED`, or `FAILED`.
</ParamField>

<ResponseField name="items" type="object[]">
  Array of transaction objects.
</ResponseField>

<ResponseField name="total" type="number">
  Total matching records.
</ResponseField>
