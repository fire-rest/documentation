---
title: "List transfers"
api: "GET https://boh.api.fire.rest/api/v1/public/vendors/{vendorId}/operations/transfers"
description: "Returns transfers for a vendor with optional pagination."
authMethod: "none"
---

<ParamField header="x-api-key" type="string" required>
  API key with `inventory:read` scope.
</ParamField>

<ParamField path="vendorId" type="string" required>
  Vendor UUID.
</ParamField>

<ParamField query="limit" type="number">
  Results per page (1–200, default 50).
</ParamField>

<ParamField query="offset" type="number">
  Pagination offset (default 0).
</ParamField>

<ParamField query="include_cancelled" type="boolean">
  Include cancelled transfers. Default `false`.
</ParamField>

<ResponseField name="items" type="object[]">
  Array of transfer objects.
</ResponseField>

<ResponseField name="total" type="number">
  Total matching records.
</ResponseField>
