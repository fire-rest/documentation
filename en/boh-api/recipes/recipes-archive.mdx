---
title: "Archive recipe"
api: "POST https://boh.api.fire.rest/api/v1/public/vendors/{vendorId}/recipes/{recipeId}/archive"
description: "Archives a recipe. BOH stops using it for consumption after archiving."
authMethod: "none"
---

<ParamField header="x-api-key" type="string" required>
  BOH API key with `recipes:write` scope.
</ParamField>

<ParamField path="vendorId" type="string" required>
  Vendor UUID.
</ParamField>

<ParamField path="recipeId" type="string" required>
  Recipe UUID.
</ParamField>

<ParamField body="external_user_id" type="string">
  Actor for audit trail.
</ParamField>

<ResponseField name="ok" type="boolean">
  `true` on success.
</ResponseField>

<RequestExample>
```json
{
  "external_user_id": "user_chef_99"
}
```
</RequestExample>

<ResponseExample>
```json 200
{
  "ok": true
}
```

```json 404
{
  "error": {
    "kind": "recipe_not_found",
    "message": "Recipe not found"
  }
}
```
</ResponseExample>
