---
title: "Unarchive item"
api: "POST https://boh.api.fire.rest/api/v1/public/vendors/{vendorId}/catalog/items/{itemId}/unarchive"
description: "Restores an archived item to active status."
authMethod: "none"
---

<ParamField header="x-api-key" type="string" required>
  BOH API key with `catalog:write` scope.
</ParamField>

<ParamField path="vendorId" type="string" required>
  Vendor UUID.
</ParamField>

<ParamField path="itemId" type="string" required>
  Item UUID.
</ParamField>

<ParamField body="external_user_id" type="string">
  Actor for audit trail.
</ParamField>

<ResponseField name="ok" type="boolean">
  `true` on success.
</ResponseField>

<RequestExample>
```json
{
  "external_user_id": "user_jane_42"
}
```
</RequestExample>

<ResponseExample>
```json 200
{
  "ok": true
}
```

```json 404
{
  "error": {
    "kind": "item_not_found",
    "message": "Item not found"
  }
}
```
</ResponseExample>
