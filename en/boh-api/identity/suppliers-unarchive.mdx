---
title: "Unarchive a supplier"
api: "POST https://boh.api.fire.rest/api/v1/public/vendors/{vendorId}/identity/suppliers/{supplierId}/unarchive"
description: "Restores an archived supplier to active status."
authMethod: "none"
---

<ParamField header="x-api-key" type="string" required>
  BOH API key with `catalog:write` scope.
</ParamField>

<ParamField path="vendorId" type="string" required>
  Vendor UUID.
</ParamField>

<ParamField path="supplierId" type="string" required>
  Supplier UUID.
</ParamField>

<ParamField body="external_user_id" type="string">
  Actor for audit trail.
</ParamField>

<ResponseField name="ok" type="boolean">
  `true` on success.
</ResponseField>

<RequestExample>
```json
{
  "external_user_id": "user_admin_01"
}
```
</RequestExample>

<ResponseExample>
```json 200
{
  "ok": true
}
```

```json 404
{
  "error": {
    "kind": "supplier_not_found",
    "message": "Supplier not found"
  }
}
```
</ResponseExample>
