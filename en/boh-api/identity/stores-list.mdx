---
title: "List stores"
api: "GET https://boh.api.fire.rest/api/v1/public/vendors/{vendorId}/identity/stores"
description: "Returns all stores (physical locations) for a vendor, with optional pagination and status filtering."
authMethod: "none"
---

<ParamField header="x-api-key" type="string" required>
  BOH API key with `catalog:read` scope.
</ParamField>

<ParamField path="vendorId" type="string" required>
  Vendor UUID.
</ParamField>

<ParamField query="limit" type="number">
  Results per page (1–200, default 50).
</ParamField>

<ParamField query="offset" type="number">
  Pagination offset (default 0).
</ParamField>

<ParamField query="status" type="string">
  Filter by `ACTIVE` or `INACTIVE`.
</ParamField>

<ResponseField name="items" type="object[]">
  Array of store objects: `id`, `account_id`, `code`, `name`, `timezone`, `status`, `external_store_id`, `tax_id`, `created_at`, `updated_at`.
</ResponseField>

<ResponseField name="total" type="number">
  Total matching records.
</ResponseField>

<ResponseField name="limit" type="number">
  Current page size.
</ResponseField>

<ResponseField name="offset" type="number">
  Current offset.
</ResponseField>
