---
title: "Submit purchase order"
api: "POST https://boh.api.fire.rest/api/v1/public/vendors/{vendorId}/procurement/orders/{orderId}?action=submit"
description: "Moves a purchase order from DRAFT to SUBMITTED status, indicating it was sent to the supplier."
authMethod: "none"
---

<ParamField header="x-api-key" type="string" required>
  API key with `inventory:write` scope.
</ParamField>

<ParamField path="vendorId" type="string" required>
  Vendor UUID.
</ParamField>

<ParamField path="orderId" type="string" required>
  Purchase order UUID. Must be in `DRAFT` status.
</ParamField>

<ParamField query="action" type="string" required>
  Must be `submit`.
</ParamField>

<ParamField body="external_user_id" type="string">
  Actor for audit trail.
</ParamField>

<ResponseField name="purchase_order" type="object">
  Updated purchase order with `status: "SUBMITTED"`.
</ResponseField>

<RequestExample>
```json
{
  "external_user_id": "user_purchasing_01"
}
```
</RequestExample>

<ResponseExample>
```json 200
{
  "purchase_order": {
    "id": "po_01j5k...",
    "status": "SUBMITTED",
    "submitted_at": "2026-08-11T13:00:00.000Z"
  }
}
```

```json 422
{
  "error": {
    "kind": "purchase_order_invalid_status_transition",
    "message": "Purchase order is not in DRAFT status"
  }
}
```
</ResponseExample>
