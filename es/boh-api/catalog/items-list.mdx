---
title: "Listar artículos"
api: "GET https://boh.api.fire.rest/api/v1/public/vendors/{vendorId}/catalog/items"
description: "Devuelve todos los artículos de catálogo para un vendor con filtros opcionales de paginación y estado."
authMethod: "none"
---

<ParamField header="x-api-key" type="string" required>
  API key de BOH con scope `catalog:read`.
</ParamField>

<ParamField path="vendorId" type="string" required>UUID del vendor.</ParamField>

<ParamField query="limit" type="number">Resultados por página (1–200, default 50).</ParamField>
<ParamField query="offset" type="number">Offset de paginación (default 0).</ParamField>
<ParamField query="status" type="string">Filtrar por `ACTIVE` o `INACTIVE`.</ParamField>

<ResponseField name="items" type="object[]">
  Array de artículos de catálogo: `id`, `account_id`, `name`, `status`, `base_unit_id`, `unit_group_id`, `external_item_id`, `metadata`, `created_at`, `updated_at`.
</ResponseField>
<ResponseField name="total" type="number">Total de registros.</ResponseField>
<ResponseField name="limit" type="number">Tamaño de página.</ResponseField>
<ResponseField name="offset" type="number">Offset actual.</ResponseField>
