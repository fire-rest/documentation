<ParamField header="x-api-key" type="string" required>

</ParamField>

<ParamField path="vendorId" type="string" required>UUID del vendor.</ParamField>

<ParamField query="store_id" type="string">Filtrar por establecimiento (origen o destino).</ParamField>
<ParamField query="from" type="string">Fecha de inicio (ISO 8601).</ParamField>
<ParamField query="to" type="string">Fecha de fin (ISO 8601).</ParamField>
<ParamField query="limit" type="number">Resultados por página (default 50).</ParamField>
<ParamField query="offset" type="number">Offset de paginación.</ParamField>

<ResponseField name="items" type="object[]">Array de transferencias.</ResponseField>
<ResponseField name="total" type="number">Total de registros.</ResponseField>
