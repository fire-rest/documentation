---
title: "Listar órdenes de compra"
api: "GET https://boh.api.fire.rest/api/v1/public/vendors/{vendorId}/procurement/orders"
description: "Devuelve las órdenes de compra del vendor con filtros opcionales."
authMethod: "none"
---

<ParamField header="x-api-key" type="string" required>
  API key de BOH con scope `inventory:read`.
</ParamField>

<ParamField path="vendorId" type="string" required>UUID del vendor.</ParamField>

<ParamField query="status" type="string">Filtrar por estado: `DRAFT`, `SUBMITTED`, `CONFIRMED`, `PARTIALLY_RECEIVED`, `CLOSED`, o `CANCELLED`.</ParamField>
<ParamField query="supplier_id" type="string">Filtrar por proveedor.</ParamField>
<ParamField query="store_id" type="string">Filtrar por establecimiento.</ParamField>
<ParamField query="from" type="string">Fecha de inicio (ISO 8601).</ParamField>
<ParamField query="to" type="string">Fecha de fin (ISO 8601).</ParamField>
<ParamField query="limit" type="number">Resultados por página (default 50).</ParamField>
<ParamField query="offset" type="number">Offset de paginación.</ParamField>

<ResponseField name="purchase_orders" type="object[]">Array de cabeceras de órdenes de compra.</ResponseField>
<ResponseField name="total" type="number">Total de registros.</ResponseField>
