---
title: "Listar establecimientos"
api: "GET https://boh.api.fire.rest/api/v1/public/vendors/{vendorId}/identity/stores"
description: "Devuelve todos los establecimientos (ubicaciones físicas) del vendor, con filtros opcionales de paginación y estado."
authMethod: "none"
---

<ParamField header="x-api-key" type="string" required>
  API key de BOH con scope `catalog:read`.
</ParamField>

<ParamField path="vendorId" type="string" required>
  UUID del vendor.
</ParamField>

<ParamField query="limit" type="number">
  Resultados por página (1–200, default 50).
</ParamField>

<ParamField query="offset" type="number">
  Offset de paginación (default 0).
</ParamField>

<ParamField query="status" type="string">
  Filtrar por `ACTIVE` o `INACTIVE`.
</ParamField>

<ResponseField name="items" type="object[]">
  Array de objetos de establecimiento: `id`, `account_id`, `code`, `name`, `timezone`, `status`, `external_store_id`, `created_at`, `updated_at`.
</ResponseField>

<ResponseField name="total" type="number">Total de registros coincidentes.</ResponseField>
<ResponseField name="limit" type="number">Tamaño de página actual.</ResponseField>
<ResponseField name="offset" type="number">Offset actual.</ResponseField>
