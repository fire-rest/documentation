---
title: "Listar canales de venta"
api: "GET https://api.fire.rest/v1/sales-channels"
description: "Devuelve los canales de venta configurados en Fire. Usa los IDs de canal al inyectar órdenes."
authMethod: "bearer"
---

<ParamField header="x-api-key" type="string" required>
  Tu API key de Fire.
</ParamField>

<ParamField query="store_id" type="string">
  Filtra los canales por tienda. Si se omite, devuelve todos los canales de todas las tiendas.
</ParamField>

<ResponseField name="channels" type="object[]">
  Lista de canales de venta.

  <Expandable title="channels[n]">
    <ResponseField name="channelId" type="string">
      Identificador único del canal. Usa este valor como `channelId` al inyectar órdenes.
    </ResponseField>
    <ResponseField name="code" type="string">
      Código corto del canal (p. ej. `IFOOD`, `RAPPI`, `KIOSK`). Identificador estable para el mapeo programático.
    </ResponseField>
    <ResponseField name="name" type="string">
      Nombre para mostrar del canal.
    </ResponseField>
    <ResponseField name="description" type="string">
      Descripción opcional del canal.
    </ResponseField>
    <ResponseField name="active" type="string">
      Estado del canal: `ACTIVE` o `INACTIVE`.
    </ResponseField>
    <ResponseField name="authType" type="string">
      Mecanismo de autenticación usado por la integración de este canal: `API_KEY`, `OAUTH2`, `BASIC_AUTH`, `XMART_LOGIN` o `NONE`.
    </ResponseField>
    <ResponseField name="baseUrl" type="string">
      URL base del endpoint de integración del canal. `null` si no aplica.
    </ResponseField>
    <ResponseField name="services" type="object[]">
      Servicios disponibles a través de este canal.

      <Expandable title="services[n]">
        <ResponseField name="code" type="string">
          Código del tipo de servicio (p. ej. `DELIVERY`, `TAKEAWAY`).
        </ResponseField>
        <ResponseField name="serviceId" type="string">
          Identificador único del servicio.
        </ResponseField>
        <ResponseField name="referenceName" type="string">
          Nombre legible del servicio para este canal.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="vendorId" type="integer">
      Identificador de la marca a la que pertenece este canal.
    </ResponseField>
    <ResponseField name="store_id" type="string">
      Tienda asociada a este canal.
    </ResponseField>
    <ResponseField name="createdAt" type="string">
      Timestamp ISO 8601 de creación del canal.
    </ResponseField>
    <ResponseField name="updatedAt" type="string">
      Timestamp ISO 8601 de la última actualización.
    </ResponseField>
  </Expandable>
</ResponseField>
