---
title: "Listar canais de venda"
api: "GET https://api.fire.rest/v1/sales-channels"
description: "Retorna os canais de venda configurados no Fire. Use os IDs de canal ao injetar pedidos."
authMethod: "bearer"
---

<ParamField header="x-api-key" type="string" required>
  Sua API key do Fire.
</ParamField>

<ParamField query="store_id" type="string">
  Filtra os canais por loja. Se omitido, retorna todos os canais de todas as lojas.
</ParamField>

<ResponseField name="channels" type="object[]">
  Lista de canais de venda.

  <Expandable title="channels[n]">
    <ResponseField name="channelId" type="string">
      Identificador único do canal. Use este valor como `channelId` ao injetar pedidos.
    </ResponseField>
    <ResponseField name="code" type="string">
      Código curto do canal (ex.: `IFOOD`, `RAPPI`, `KIOSK`). Identificador estável para mapeamento programático.
    </ResponseField>
    <ResponseField name="name" type="string">
      Nome de exibição do canal.
    </ResponseField>
    <ResponseField name="description" type="string">
      Descrição opcional do canal.
    </ResponseField>
    <ResponseField name="active" type="string">
      Status do canal: `ACTIVE` ou `INACTIVE`.
    </ResponseField>
    <ResponseField name="authType" type="string">
      Mecanismo de autenticação usado pela integração deste canal: `API_KEY`, `OAUTH2`, `BASIC_AUTH`, `XMART_LOGIN` ou `NONE`.
    </ResponseField>
    <ResponseField name="baseUrl" type="string">
      URL base do endpoint de integração do canal. `null` se não se aplicar.
    </ResponseField>
    <ResponseField name="services" type="object[]">
      Serviços disponíveis através deste canal.

      <Expandable title="services[n]">
        <ResponseField name="code" type="string">
          Código do tipo de serviço (ex.: `DELIVERY`, `TAKEAWAY`).
        </ResponseField>
        <ResponseField name="serviceId" type="string">
          Identificador único do serviço.
        </ResponseField>
        <ResponseField name="referenceName" type="string">
          Nome legível do serviço para este canal.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="vendorId" type="integer">
      Identificador da marca à qual este canal pertence.
    </ResponseField>
    <ResponseField name="store_id" type="string">
      Loja associada a este canal.
    </ResponseField>
    <ResponseField name="createdAt" type="string">
      Timestamp ISO 8601 de criação do canal.
    </ResponseField>
    <ResponseField name="updatedAt" type="string">
      Timestamp ISO 8601 da última atualização.
    </ResponseField>
  </Expandable>
</ResponseField>
