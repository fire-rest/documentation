---
title: "Reportar evento fiscal"
api: "POST https://api.fire.rest/v1/webhooks/fiscal/callback"
description: "Reporte o resultado da sua integração fiscal local ao Fire (CO, EC, CL, AR, VE)."
authMethod: "bearer"
---

Endpoint único multipaís para reportar eventos fiscais. Para o contexto conceitual, quando chamar e exemplos por país, veja o [guia do callback fiscal](/pt/configuration/fiscal-callback).

### Headers

<ParamField header="x-api-key" type="string" required>
  API key do Fire com o scope `webhooks:fiscal`.
</ParamField>

### Body

<ParamField body="countryCode" type="string" required>
  Código ISO de país. Um de `CO`, `EC`, `CL`, `AR`, `VE`. Define a validação de `document`.
</ParamField>

<ParamField body="eventType" type="string" required>
  Um de `authorized`, `cancelled`, `rejected`, `denied`, `error`.
</ParamField>

<ParamField body="providerEventId" type="string" required>
  Chave de idempotência. Fire indexa `(provider_code, providerEventId)` UNIQUE.
</ParamField>

<ParamField body="occurredAt" type="string" required>
  Timestamp ISO 8601 UTC de quando o evento ocorreu no seu sistema.
</ParamField>

<ParamField body="orderId" type="string" required>
  UUID do pedido no Fire — `data.orderId` do envelope que o Fire emitiu.
</ParamField>

<ParamField body="eventId" type="string" required>
  O `event.id` que o Fire enviou no envelope original. Correlaciona o callback com a execução específica.
</ParamField>

<ParamField body="document" type="object">
  Dados fiscais por país. Obrigatório quando `eventType` é `authorized` ou `cancelled`.

  <Expandable title="propriedades">
    <ParamField body="docType" type="string" required>
      `invoice` ou `cancellation`.
    </ParamField>
    <ParamField body="docSubtype" type="string" required>
      Subtipo do país (`factura`, `boleta`, etc.).
    </ParamField>
    <ParamField body="totalAmount" type="number" required>
      Valor total com impostos.
    </ParamField>
    <ParamField body="taxAmount" type="number">
      Parcela de impostos do total.
    </ParamField>
    <ParamField body="currencyCode" type="string" required>
      ISO 4217 (`COP`, `USD`, `CLP`, `ARS`, `VES`).
    </ParamField>
    <ParamField body="pdfUrl" type="string">
      URL onde o PDF do documento legal está hospedado.
    </ParamField>
    <ParamField body="xmlUrl" type="string">
      URL onde o XML do documento legal está hospedado.
    </ParamField>
    <ParamField body="emittedAt" type="string">
      ISO 8601 UTC.
    </ParamField>
    <ParamField body="cancelledAt" type="string">
      ISO 8601 UTC. Obrigatório quando `eventType=cancelled`.
    </ParamField>
    <ParamField body="cufe" type="string">

    <ParamField body="prefijo" type="string">

    <ParamField body="numeroDian" type="string">

    <ParamField body="qrCode" type="string">

    </ParamField>
    <ParamField body="claveAcceso" type="string">
      🇪🇨 Equador — chave de acesso SRI de 49 caracteres.
    </ParamField>
    <ParamField body="numeroAutorizacion" type="string">
      🇪🇨 Equador.
    </ParamField>
    <ParamField body="ambiente" type="string">
      🇪🇨 Equador — `1` (produção) ou `2` (testes).
    </ParamField>
    <ParamField body="folio" type="string">
      🇨🇱 Chile — folio SII.
    </ParamField>
    <ParamField body="ted" type="string">
      🇨🇱 Chile — string TED PKCS#7.
    </ParamField>
    <ParamField body="tipoDte" type="string">
      🇨🇱 Chile.
    </ParamField>
    <ParamField body="trackId" type="string">
      🇨🇱 Chile.

    <ParamField body="cae" type="string">
      🇦🇷 Argentina — CAE de 14 dígitos.
    </ParamField>
    <ParamField body="fechaVtoCae" type="string">
      🇦🇷 Argentina — `YYYY-MM-DD`.

  </Expandable>

<ParamField body="error" type="object">
  Obrigatório quando `eventType` é `rejected`, `denied` ou `error`.

  <Expandable title="propriedades">
    <ParamField body="code" type="string">
      Código de erro específico do provedor.
    </ParamField>
    <ParamField body="message" type="string">
      Descrição legível para humanos.
    </ParamField>
  </Expandable>
</ParamField>

### Resposta

<ResponseField name="received" type="boolean">
  Sempre `true` em `2xx`.
</ResponseField>

<ResponseField name="receiptId" type="string">
  Identificador interno do Fire para esta entrega.
</ResponseField>

<ResponseField name="outcome" type="string">
  Um de `created`, `updated`, `idempotent`, `regression`, `duplicate`.
</ResponseField>

<ResponseField name="documentId" type="string">
  ID interno do Fire do documento fiscal persistido.
</ResponseField>

<ResponseField name="processingMs" type="number">
  Tempo de processamento server-side em milissegundos.
</ResponseField>
