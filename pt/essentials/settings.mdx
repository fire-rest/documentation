---
title: 'Configurações globais'
description: 'O Mintlify oferece controle total sobre a aparência da sua documentação com o arquivo docs.json'
icon: 'gear'
---

Todo site Mintlify precisa de um arquivo `docs.json` com as configurações principais. Saiba mais sobre as [propriedades](#properties) abaixo.

## Propriedades

<ResponseField name="name" type="string" required>
Nome do projeto. Usado como título global.

Exemplo: `mintlify`

</ResponseField>

<ResponseField name="navigation" type="Navigation[]" required>
  Lista de grupos com as páginas de cada grupo
  <Expandable title="Navigation">
    <ResponseField name="group" type="string">
    Nome do grupo.

    Exemplo: `Settings`

    </ResponseField>
    <ResponseField name="pages" type="string[]">
    Caminhos relativos aos arquivos markdown que serão páginas.

    Exemplo: `["customization", "page"]`

    </ResponseField>

  </Expandable>
</ResponseField>

<ResponseField name="logo" type="string or object">
  Caminho do logo ou objeto com caminhos para modo claro e escuro
  <Expandable title="Logo">
    <ResponseField name="light" type="string">
      Caminho do logo no modo claro
    </ResponseField>
    <ResponseField name="dark" type="string">
      Caminho do logo no modo escuro
    </ResponseField>
    <ResponseField name="href" type="string" default="/">
      Destino ao clicar no logo
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="favicon" type="string">
  Caminho do favicon
</ResponseField>

<ResponseField name="colors" type="Colors">
  Cores hex do tema global
  <Expandable title="Colors">
    <ResponseField name="primary" type="string" required>
      Cor primária: conteúdo em destaque, cabeçalhos de seção, acentos no modo claro
    </ResponseField>
    <ResponseField name="light" type="string">
      Cor primária no modo escuro: conteúdo em destaque, cabeçalhos, acentos
    </ResponseField>
    <ResponseField name="dark" type="string">
      Cor primária para botões importantes
    </ResponseField>
    <ResponseField name="background" type="object">
      Cor de fundo nos modos claro e escuro
      <Expandable title="Object">
        <ResponseField name="light" type="string" required>
          Código hex do fundo no modo claro
        </ResponseField>
        <ResponseField name="dark" type="string" required>
          Código hex do fundo no modo escuro
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="topbarLinks" type="TopbarLink[]">
  Lista de `name` e `url` dos links na barra superior
  <Expandable title="TopbarLink">
    <ResponseField name="name" type="string">
    Texto do botão.

    Exemplo: `Contact us`
    </ResponseField>
    <ResponseField name="url" type="string">
    URL ao clicar. Exemplo: `https://mintlify.com/docs`
    </ResponseField>

  </Expandable>
</ResponseField>

<ResponseField name="topbarCtaButton" type="Call to Action">
  <Expandable title="Topbar Call to Action">
    <ResponseField name="type" type={'"link" or "github"'} default="link">
    Com `link` exibe um botão. Com `github` exibe informações do repositório e estrelas.
    </ResponseField>
    <ResponseField name="url" type="string">
    Se for `link`: destino do botão.
    
    Se for `github`: URL do repositório para carregar dados do GitHub.
    </ResponseField>
    <ResponseField name="name" type="string">
    Texto do botão. Obrigatório apenas se `type` for `link`.
    </ResponseField>

  </Expandable>
</ResponseField>

<ResponseField name="versions" type="string[]">
  Nomes de versão. Use apenas se quiser várias versões da documentação com seletor na barra de navegação.
</ResponseField>

<ResponseField name="anchors" type="Anchor[]">
  Lista de âncoras com `icon`, `color` e `url`.
  <Expandable title="Anchor">
    <ResponseField name="icon" type="string">
    Ícone [Font Awesome](https://fontawesome.com/search?q=heart) da âncora.

    Exemplo: `comments`
    </ResponseField>
    <ResponseField name="name" type="string">
    Rótulo da âncora.

    Exemplo: `Community`
    </ResponseField>
    <ResponseField name="url" type="string">
      Prefixo de URL das páginas desta âncora. Geralmente coincide com o nome da pasta de páginas.
    </ResponseField>
    <ResponseField name="color" type="string">
      Cor hex do fundo do ícone. Também pode ser um gradiente com `from` e `to` em hex.
    </ResponseField>
    <ResponseField name="version" type="string">
      Use se quiser ocultar a âncora até que a versão correta da documentação seja selecionada.
    </ResponseField>
    <ResponseField name="isDefaultHidden" type="boolean" default="false">
      `true` para ocultar a âncora até que alguém link diretamente para documentação dentro dela.
    </ResponseField>
    <ResponseField name="iconType" default="duotone" type="string">
      Um de: "brands", "duotone", "light", "sharp-solid", "solid" ou "thin"
    </ResponseField>

  </Expandable>
</ResponseField>

<ResponseField name="topAnchor" type="Object">
  Substitui a configuração da âncora superior.
  <Expandable title="Object">
    <ResponseField name="name" default="Documentation" type="string">
      Nome da âncora superior
    </ResponseField>
    <ResponseField name="icon" default="book-open" type="string">
      Ícone Font Awesome.
    </ResponseField>
    <ResponseField name="iconType" default="duotone" type="string">
      Um de: "brands", "duotone", "light", "sharp-solid", "solid" ou "thin"
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="tabs" type="Tabs[]">
  Abas de navegação principais.
  <Expandable title="Tabs">
    <ResponseField name="name" type="string">
      Rótulo da aba.
    </ResponseField>
    <ResponseField name="url" type="string">
      Prefixo de URL das páginas desta aba. Geralmente coincide com o nome da pasta.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="api" type="API">
  Configuração da API. Mais informações em [componentes de API](https://mintlify.com/docs/api-playground/mdx/configuration).
  <Expandable title="API">
    <ResponseField name="baseUrl" type="string">
      URL base dos endpoints. Se `baseUrl` for um array, o usuário pode alternar entre várias bases.
    </ResponseField>

    <ResponseField name="auth" type="Auth">
      <Expandable title="Auth">
        <ResponseField name="method" type='"bearer" | "basic" | "key"'>
          Estratégia de autenticação para todos os endpoints.
        </ResponseField>
        <ResponseField name="name" type="string">
        Nome do parâmetro de autenticação no playground.

        Se o método for `basic`, o formato deve ser `[usuário]:[senha]`
        </ResponseField>
        <ResponseField name="inputPrefix" type="string">
        Prefixo padrão do campo de autenticação.

        Ex.: com `inputPrefix` `AuthKey`, o valor padrão do campo será `AuthKey`.
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="playground" type="Playground">
      Opções do playground de API

      <Expandable title="Playground">
        <ResponseField name="mode" default="show" type='"show" | "simple" | "hide"'>
          Se o playground está visível, oculto ou em modo `simple` (apenas o endpoint, sem interação).

          Mais nas [guias do playground](https://mintlify.com/docs/api-playground/overview)
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="maintainOrder" type="boolean">
      Garante que a ordem das chaves nas páginas OpenAPI corresponda ao arquivo OpenAPI.

      <Warning>Em breve será o comportamento padrão e este campo será descontinuado.</Warning>
    </ResponseField>

  </Expandable>
</ResponseField>

<ResponseField name="openapi" type="string | string[]">
  String ou lista de URLs ou caminhos relativos ao arquivo OpenAPI.
  
  Exemplos:
  <CodeGroup>
    ```json Absolute
    "openapi": "https://example.com/openapi.json"
    ```
    ```json Relative
    "openapi": "/openapi.json"
    ```
    ```json Multiple
    "openapi": ["https://example.com/openapi1.json", "/openapi2.json", "/openapi3.json"]
    ```
  </CodeGroup>

</ResponseField>

<ResponseField name="footerSocials" type="FooterSocials">
  Objeto de redes sociais: cada chave é a plataforma e o valor é a URL.
  
  Exemplo: 
  ```json
  {
    "x": "https://x.com/mintlify",
    "website": "https://mintlify.com"
  }
  ```
  <Expandable title="FooterSocials">
    <ResponseField name="[key]" type="string">
    Uma de: `website`, `facebook`, `x`, `discord`, `slack`, `github`, `linkedin`, `instagram`, `hacker-news`
    
    Exemplo: `x`
    </ResponseField>
    <ResponseField name="property" type="string">
    URL do perfil ou página.
    
    Exemplo: `https://x.com/mintlify`
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="feedback" type="Feedback">
  Botões de feedback

  <Expandable title="Feedback">
    <ResponseField name="suggestEdit" type="boolean" default="false">
    Botão para sugerir edições via pull requests
    </ResponseField>
    <ResponseField name="raiseIssue" type="boolean" default="false">
    Botão para abrir um issue sobre a documentação
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="modeToggle" type="ModeToggle">
  Personalize o alternador claro/escuro.
  <Expandable title="ModeToggle">
    <ResponseField name="default" type={'"light" or "dark"'}>
      Força modo claro ou escuro para novos usuários. Se não definido, usa a preferência do sistema.
    </ResponseField>
    <ResponseField name="isHidden" type="boolean" default="false">
      `true` para ocultar o alternador. Combine com `default` para forçar apenas claro ou escuro. Por exemplo:
      
      <CodeGroup>
      ```json Only Dark Mode
      "modeToggle": {
        "default": "dark",
        "isHidden": true
      }
      ```

      ```json Only Light Mode
      "modeToggle": {
        "default": "light",
        "isHidden": true
      }
      ```
      </CodeGroup>

    </ResponseField>

  </Expandable>
</ResponseField>

<ResponseField name="backgroundImage" type="string">
  Imagem de fundo atrás de todas as páginas. Exemplos:
  [Infisical](https://infisical.com/docs) e [FRPC](https://frpc.io).
</ResponseField>
