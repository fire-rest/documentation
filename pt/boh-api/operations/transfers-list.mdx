authMethod: "none"
---

<ParamField header="x-api-key" type="string" required>
  API key do BOH com escopo `inventory:read`.
</ParamField>

<ParamField path="vendorId" type="string" required>UUID do vendor.</ParamField>
<ParamField query="store_id" type="string">Filtrar por estabelecimento (origem ou destino).</ParamField>
<ParamField query="from" type="string">Data de início (ISO 8601).</ParamField>
<ParamField query="to" type="string">Data de fim (ISO 8601).</ParamField>
<ParamField query="limit" type="number">Resultados por página (padrão 50).</ParamField>
<ParamField query="offset" type="number">Offset de paginação.</ParamField>

<ResponseField name="items" type="object[]">Array de transferências.</ResponseField>
<ResponseField name="total" type="number">Total de registros.</ResponseField>
