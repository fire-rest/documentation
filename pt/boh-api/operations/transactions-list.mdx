</ParamField>

<ParamField query="type" type="string">Filtrar por tipo de transação.</ParamField>
<ParamField query="store_id" type="string">Filtrar por estabelecimento.</ParamField>
<ParamField query="from" type="string">Data de início (ISO 8601).</ParamField>
<ParamField query="to" type="string">Data de fim (ISO 8601).</ParamField>
<ParamField query="limit" type="number">Resultados por página (padrão 50).</ParamField>
<ParamField query="offset" type="number">Offset de paginação.</ParamField>

<ResponseField name="items" type="object[]">Array de transações.</ResponseField>
<ResponseField name="total" type="number">Total de registros.</ResponseField>
