---
title: "Adicionar linhas ao pedido de compra"
api: "POST https://boh.api.fire.rest/api/v1/public/vendors/{vendorId}/procurement/orders/{orderId}?action=add-lines"
description: "Adiciona novas linhas a um pedido de compra existente em status DRAFT ou CONFIRMED."
authMethod: "none"
---

<ParamField header="x-api-key" type="string" required>
  API key do BOH com escopo `inventory:write`.
</ParamField>

<ParamField path="vendorId" type="string" required>UUID do vendor.</ParamField>
<ParamField path="orderId" type="string" required>UUID do pedido de compra.</ParamField>

<ParamField body="lines" type="object[]" required>
  <Expandable title="campos da linha">
    <ParamField body="item_id" type="string">UUID do item.</ParamField>
    <ParamField body="external_item_id" type="string">ID externo do item.</ParamField>
    <ParamField body="quantity" type="number" required>Quantidade a pedir.</ParamField>
    <ParamField body="unit_id" type="string" required>UUID da unidade.</ParamField>
    <ParamField body="unit_cost" type="number">Custo unitário.</ParamField>
  </Expandable>
</ParamField>
<ParamField body="external_user_id" type="string" required>Ator que adiciona linhas.</ParamField>

<RequestExample>
```json
{
  "lines": [
    {
      "external_item_id": "cebola",
      "quantity": 10,
      "unit_id": "unit_kg_01..."
    }
  ],
  "external_user_id": "compras_01"
}
```
</RequestExample>

<ResponseExample>
```json 200
{ "purchase_order": { "id": "po_01...", "status": "DRAFT" } }
```
</ResponseExample>
